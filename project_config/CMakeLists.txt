string(TOUPPER ${%%CPP_INIT_REPLACE%%_PROJECT_NAME} %%CPP_INIT_REPLACE%%_PROJECT_NAME_CAPS)
string(REPLACE "-" "_" %%CPP_INIT_REPLACE%%_SYSTEM_NAME_SANITISED ${CMAKE_SYSTEM_NAME})
string(REPLACE "." ";" _%%cpp_init_replace%%_compiler_version_list ${CMAKE_CXX_COMPILER_VERSION})
list(LENGTH _%%cpp_init_replace%%_compiler_version_list _%%cpp_init_replace%%_compiler_version_len)
# TODO(dk949): This will break if compiler versions exceed 99
if (_%%cpp_init_replace%%_compiler_version_len GREATER_EQUAL 4)
    list(GET _%%cpp_init_replace%%_compiler_version_list 3 %%CPP_INIT_REPLACE%%_COMPILER_VERSION_TWEAK)
    string(REGEX REPLACE "0(.)" "\\1" %%CPP_INIT_REPLACE%%_COMPILER_VERSION_TWEAK ${%%CPP_INIT_REPLACE%%_COMPILER_VERSION_TWEAK})
else ()
    set(%%CPP_INIT_REPLACE%%_COMPILER_VERSION_TWEAK 0)
endif ()
if (_%%cpp_init_replace%%_compiler_version_len GREATER_EQUAL 3)
    list(GET _%%cpp_init_replace%%_compiler_version_list 2 %%CPP_INIT_REPLACE%%_COMPILER_VERSION_PATCH)
    string(REGEX REPLACE "0(.)" "\\1" %%CPP_INIT_REPLACE%%_COMPILER_VERSION_PATCH ${%%CPP_INIT_REPLACE%%_COMPILER_VERSION_PATCH})
else ()
    set(%%CPP_INIT_REPLACE%%_COMPILER_VERSION_PATCH 0)
endif ()
if (_%%cpp_init_replace%%_compiler_version_len GREATER_EQUAL 2)
    list(GET _%%cpp_init_replace%%_compiler_version_list 1 %%CPP_INIT_REPLACE%%_COMPILER_VERSION_MINOR)
    string(REGEX REPLACE "0(.)" "\\1" %%CPP_INIT_REPLACE%%_COMPILER_VERSION_MINOR ${%%CPP_INIT_REPLACE%%_COMPILER_VERSION_MINOR})
else ()
    set(%%CPP_INIT_REPLACE%%_COMPILER_VERSION_MINOR 0)
endif ()
if (NOT (CMAKE_CXX_COMPILER_VERSION STREQUAL ""))
    list(GET _%%cpp_init_replace%%_compiler_version_list 0 %%CPP_INIT_REPLACE%%_COMPILER_VERSION_MAJOR)
    string(REGEX REPLACE "0(.)" "\\1" %%CPP_INIT_REPLACE%%_COMPILER_VERSION_MAJOR ${%%CPP_INIT_REPLACE%%_COMPILER_VERSION_MAJOR})
else ()
    set(%%CPP_INIT_REPLACE%%_COMPILER_VERSION_MAJOR 0)
endif ()

configure_file("config.hpp.in" "${PROJECT_BINARY_DIR}/project_config/include/project/config.hpp" ESCAPE_QUOTES)
